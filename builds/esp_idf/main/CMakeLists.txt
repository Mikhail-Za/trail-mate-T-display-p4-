target_compile_definitions(${COMPONENT_LIB} PRIVATE
    TRAIL_MATE_LORA_TX_POWER_MAX_DBM=22)

# RadioLib: trim to SX1262 only -- disable every other module/protocol family so
# their headers stay inert (the source list already omits their .cpp's). Do NOT
# add RADIOLIB_BUILD_ARDUINO or the Arduino-as-component to REQUIRES, or RadioLib
# flips to the Arduino path and demands <SPI.h>/Arduino.h.
target_compile_definitions(${COMPONENT_LIB} PRIVATE
    RADIOLIB_EXCLUDE_CC1101 RADIOLIB_EXCLUDE_NRF24 RADIOLIB_EXCLUDE_RF69
    RADIOLIB_EXCLUDE_SI443X RADIOLIB_EXCLUDE_SX127X RADIOLIB_EXCLUDE_SX128X
    RADIOLIB_EXCLUDE_LR11X0 RADIOLIB_EXCLUDE_LR2021 RADIOLIB_EXCLUDE_AFSK
    RADIOLIB_EXCLUDE_RTTY RADIOLIB_EXCLUDE_MORSE RADIOLIB_EXCLUDE_HELLSCHREIBER
    RADIOLIB_EXCLUDE_SSTV RADIOLIB_EXCLUDE_AX25 RADIOLIB_EXCLUDE_FSK4
    RADIOLIB_EXCLUDE_PAGER RADIOLIB_EXCLUDE_APRS RADIOLIB_EXCLUDE_BELL
    RADIOLIB_EXCLUDE_LORAWAN RADIOLIB_EXCLUDE_DIRECT_RECEIVE)

if(TRAIL_MATE_IDF_TARGET STREQUAL "tab5")
    target_compile_definitions(${COMPONENT_LIB} PRIVATE TRAIL_MATE_ESP_BOARD_TAB5=1)
elseif(TRAIL_MATE_IDF_TARGET STREQUAL "tdisplayp4_tft")
    target_compile_definitions(${COMPONENT_LIB} PRIVATE TRAIL_MATE_ESP_BOARD_T_DISPLAY_P4=1)
elseif(TRAIL_MATE_IDF_TARGET STREQUAL "tdisplayp4_amoled")
    target_compile_definitions(${COMPONENT_LIB} PRIVATE
        TRAIL_MATE_ESP_BOARD_T_DISPLAY_P4=1
        TRAIL_MATE_ESP_BOARD_T_DISPLAY_P4_AMOLED=1)
endif()
